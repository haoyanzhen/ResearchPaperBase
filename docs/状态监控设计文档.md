### 3.2 指标采集器

```python
import psutil
import time
from threading import Thread
from typing import Callable

class MetricsCollector:
    """指标采集器"""

    def __init__(self):
        self.system_metrics_callbacks = []
        self.application_metrics_callbacks = []
        self.business_metrics_callbacks = []

    def start_system_metrics_collection(self, interval: int = 60):
        """启动系统指标采集"""
        def collect():
            while True:
                metrics = self._collect_system_metrics()
                for callback in self.system_metrics_callbacks:
                    callback(metrics)
                time.sleep(interval)

        thread = Thread(target=collect, daemon=True)
        thread.start()

    def start_application_metrics_collection(self, interval: int = 30):
        """启动应用指标采集"""
        def collect():
            while True:
                metrics = self._collect_application_metrics()
                for callback in self.application_metrics_callbacks:
                    callback(metrics)
                time.sleep(interval)

        thread = Thread(target=collect, daemon=True)
        thread.start()

    def start_business_metrics_collection(self, interval: int = 300):
        """启动业务指标采集"""
        def collect():
            while True:
                metrics = self._collect_business_metrics()
                for callback in self.business_metrics_callbacks:
                    callback(metrics)
                time.sleep(interval)

        thread = Thread(target=collect, daemon=True)
        thread.start()

    def register_system_metrics_callback(self, callback: Callable):
        """注册系统指标回调"""
        self.system_metrics_callbacks.append(callback)

    def register_application_metrics_callback(self, callback: Callable):
        """注册应用指标回调"""
        self.application_metrics_callbacks.append(callback)

    def register_business_metrics_callback(self, callback: Callable):
        """注册业务指标回调"""
        self.business_metrics_callbacks.append(callback)

    def _collect_system_metrics(self) -> SystemMetrics:
        """采集系统指标"""
        # CPU指标
        cpu_usage_percent = psutil.cpu_percent(interval=1)
        cpu_core_count = psutil.cpu_count()

        # 内存指标
        memory = psutil.virtual_memory()
        memory_usage_percent = memory.percent
        memory_total_gb = memory.total / (1024 ** 3)
        memory_available_gb = memory.available / (1024 ** 3)

        # 磁盘指标
        disk = psutil.disk_usage('/')
        disk_usage_percent = disk.percent
        disk_total_gb = disk.total / (1024 ** 3)
        disk_available_gb = disk.available / (1024 ** 3)

        # 网络指标
        network = psutil.net_io_counters()
        network_sent_mb = network.bytes_sent / (1024 ** 2)
        network_received_mb = network.bytes_recv / (1024 ** 2)

        # 系统负载
        load_avg = psutil.getloadavg()
        load_average_1m = load_avg[0]
        load_average_5m = load_avg[1]
        load_average_15m = load_avg[2]

        return SystemMetrics(
            cpu_usage_percent=cpu_usage_percent,
            cpu_core_count=cpu_core_count,
            memory_usage_percent=memory_usage_percent,
            memory_total_gb=memory_total_gb,
            memory_available_gb=memory_available_gb,
            disk_usage_percent=disk_usage_percent,
            disk_total_gb=disk_total_gb,
            disk_available_gb=disk_available_gb,
            network_sent_mb=network_sent_mb,
            network_received_mb=network_received_mb,
            load_average_1m=load_average_1m,
            load_average_5m=load_average_5m,
            load_average_15m=load_average_15m,
            timestamp=datetime.now()
        )

    def _collect_application_metrics(self) -> ApplicationMetrics:
        """采集应用指标"""
        # 这里需要从应用的实际数据源采集
        # 例如：从数据库、日志、计数器等

        return ApplicationMetrics(
            request_count=0,
            request_success_count=0,
            request_error_count=0,
            request_avg_response_time_ms=0.0,
            db_query_count=0,
            db_query_avg_time_ms=0.0,
            db_connection_count=0,
            llm_call_count=0,
            llm_call_success_count=0,
            llm_call_error_count=0,
            llm_call_avg_time_ms=0.0,
            api_call_count=0,
            api_call_success_count=0,
            api_call_error_count=0,
            api_call_avg_time_ms=0.0,
            task_total_count=0,
            task_running_count=0,
            task_completed_count=0,
            task_failed_count=0,
            timestamp=datetime.now()
        )

    def _collect_business_metrics(self) -> BusinessMetrics:
        """采集业务指标"""
        # 这里需要从数据库采集业务数据

        return BusinessMetrics(
            active_user_count=0,
            new_user_count=0,
            total_project_count=0,
            active_project_count=0,
            construction_mode_project_count=0,
            deep_research_mode_project_count=0,
```

### 3.3 指标存储

```python
import sqlite3
from datetime import datetime, timedelta
from typing import List, Dict, Optional

class MetricsStorage:
    """指标存储"""

    def __init__(self, db_path: str = "monitoring.db"):
        self.db_path = db_path
        self._initialize_database()

    def _initialize_database(self):
        """初始化数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # 创建系统指标表
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS system_metrics (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp DATETIME NOT NULL,
                cpu_usage_percent REAL,
                memory_usage_percent REAL,
                disk_usage_percent REAL,
                load_average_1m REAL
            )
        """)

        # 创建应用指标表
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS application_metrics (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp DATETIME NOT NULL,
                request_count INTEGER,
                request_error_count INTEGER,
                request_avg_response_time_ms REAL,
                llm_call_count INTEGER,
                llm_call_error_count INTEGER,
                llm_call_avg_time_ms REAL
            )
        """)

        # 创建业务指标表
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS business_metrics (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp DATETIME NOT NULL,
                active_user_count INTEGER,
                total_project_count INTEGER,
                total_paper_count INTEGER
            )
        """)

        conn.commit()
        conn.close()

    def save_system_metrics(self, metrics: SystemMetrics):
        """保存系统指标"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute("""
            INSERT INTO system_metrics (
                timestamp, cpu_usage_percent, memory_usage_percent,
                disk_usage_percent, load_average_1m
            ) VALUES (?, ?, ?, ?, ?)
        """, (
            metrics.timestamp,
            metrics.cpu_usage_percent,
            metrics.memory_usage_percent,
            metrics.disk_usage_percent,
            metrics.load_average_1m
        ))

        conn.commit()
        conn.close()

    def save_application_metrics(self, metrics: ApplicationMetrics):
        """保存应用指标"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        cursor.execute("""
            INSERT INTO application_metrics (
                timestamp, request_count, request_error_count,
                request_avg_response_time_ms, llm_call_count,
                llm_call_error_count, llm_call_avg_time_ms
            ) VALUES (?, ?, ?, ?, ?, ?, ?)

            metrics.request_avg_response_time_ms,
            metrics.llm_call_count,
            metrics.llm_call_error_count,

        conn.close()

        cursor.execute("""
            INSERT INTO business_metrics (
                timestamp, active_user_count, total_project_count,
                total_paper_count
            ) VALUES (?, ?, ?, ?)
        """, (
            metrics.timestamp,
            metrics.active_user_count,
            metrics.total_project_count,
            metrics.total_paper_count
        ))

        conn.commit()
        conn.close()

    def get_system_metrics(self, time_range: str = "1h") -> List[Dict]:
        """获取系统指标"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # 计算时间范围
        end_time = datetime.now()
        if time_range == "1h":
            start_time = end_time - timedelta(hours=1)
        elif time_range == "24h":
            start_time = end_time - timedelta(hours=24)
        elif time_range == "7d":

            SELECT * FROM system_metrics
```

### 4.1 告警规则定义

```python
from enum import Enum
from dataclasses import dataclass

class AlertSeverity(Enum):
    """告警严重程度"""
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"

@dataclass
class AlertRule:
    """告警规则"""
    name: str
    metric_type: str  # system, application, business
    metric_name: str
    condition: str  # >, <, >=, <=, ==
    threshold: float
    severity: AlertSeverity
    duration: int  # 持续时间（秒）
    enabled: bool = True

@dataclass
class Alert:
    """告警"""
    rule_name: str
    metric_type: str
    metric_name: str
    current_value: float
    threshold: float
    severity: AlertSeverity
    message: str
    timestamp: datetime

    def to_dict(self) -> Dict:
        """转换为字典"""
        return {
            'rule_name': self.rule_name,
            'metric_type': self.metric_type,
            'metric_name': self.metric_name,
            'current_value': self.current_value,
            'threshold': self.threshold,
            'severity': self.severity.value,
            'message': self.message,
            'timestamp': self.timestamp.isoformat()
        }
```

```python
class DashboardGenerator:
    """仪表盘数据生成器"""

    def __init__(self, monitoring_system: MonitoringSystem):
        self.monitoring_system = monitoring_system

    def generate_dashboard_data(self) -> Dict:
        """生成仪表盘数据"""
        return {
            'system_health': self._get_system_health(),
            'application_health': self._get_application_health(),
            'business_overview': self._get_business_overview(),
            'recent_alerts': self._get_recent_alerts(),
            'metrics_trends': self._get_metrics_trends()
        }

    def _get_system_health(self) -> Dict:
        """获取系统健康度"""
        system_metrics = self.monitoring_system.get_system_metrics("1h")

        if not system_metrics:
            return {
                'status': 'unknown',
                'message': '无法获取系统指标'
            }

        latest_metrics = system_metrics[0]

        # 计算系统健康度
        if (latest_metrics['cpu_usage_percent'] < 80 and
            latest_metrics['memory_usage_percent'] < 80 and
            latest_metrics['disk_usage_percent'] < 85):
            return {
                'status': 'healthy',
                'message': '系统运行正常'
            }
        elif (latest_metrics['cpu_usage_percent'] < 90 and
              latest_metrics['memory_usage_percent'] < 90):
            return {
                'status': 'warning',
                'message': '系统资源使用较高'
            }
        else:
            return {
                'status': 'critical',
                'message': '系统资源使用过高'
            }

    def _get_application_health(self) -> Dict:
        """获取应用健康度"""
        application_metrics = self.monitoring_system.get_application_metrics("1h")

        if not application_metrics:
            return {
                'status': 'unknown',
                'message': '无法获取应用指标'
            }

        latest_metrics = application_metrics[0]

        # 计算应用健康度
        request_error_rate = (
            latest_metrics['request_error_count'] /
            latest_metrics['request_count']
            if latest_metrics['request_count'] > 0
            else 0
        )

        if request_error_rate < 0.05:
            return {
                'status': 'healthy',
                'message': '应用运行正常'
            }
        elif request_error_rate < 0.1:
            return {
                'status': 'warning',
                'message': '应用错误率较高'
            }
        else:
            return {
                'status': 'critical',
                'message': '应用错误率过高'
            }

    def _get_business_overview(self) -> Dict:
        """获取业务概览"""
        business_metrics = self.monitoring_system.get_business_metrics("24h")

        if not business_metrics:
            return {}

        latest_metrics = business_metrics[0]

        return {
            'active_users': latest_metrics['active_user_count'],
            'total_projects': latest_metrics['total_project_count'],
            'total_papers': latest_metrics['total_paper_count']
        }

    def _get_recent_alerts(self) -> List[Dict]:
        """获取最近的告警"""
        alert_manager = self.monitoring_system.alert_manager
        alerts = alert_manager.get_alert_history("24h")

        return [alert.to_dict() for alert in alerts[:10]]

    def _get_metrics_trends(self) -> Dict:
        """获取指标趋势"""
        return {
            'cpu_usage': self._get_metrics_trend('cpu_usage_percent', '1h'),
            'memory_usage': self._get_metrics_trend('memory_usage_percent', '1h'),
            'request_count': self._get_application_metrics_trend('request_count', '1h'),
            'llm_call_count': self._get_application_metrics_trend('llm_call_count', '1h')
        }

    def _get_metrics_trend(self, metric_name: str, time_range: str) -> List[Dict]:
        """获取指标趋势"""
        metrics = self.monitoring_system.get_system_metrics(time_range)

        return [
            {
                'timestamp': metric['timestamp'],
                'value': metric[metric_name]
            }
            for metric in metrics
        ]

    def _get_application_metrics_trend(self, metric_name: str, time_range: str) -> List[Dict]:
        """获取应用指标趋势"""
        metrics = self.monitoring_system.get_application_metrics(time_range)

        return [
            {
                'timestamp': metric['timestamp'],
                'value': metric[metric_name]
            }
            for metric in metrics
        ]
```
