## 3. 错误处理机制

### 3.1 全局异常处理器

```python
from typing import Dict, Any
from datetime import datetime
import traceback

class ErrorResponse:
    """错误响应"""

    def __init__(self, error_type: ErrorType, error_code: int, message: str,
                 severity: ErrorSeverity, details: Dict[str, Any] = None):
        self.error_type = error_type
        self.error_code = error_code
        self.message = message
        self.severity = severity
        self.details = details or {}
        self.timestamp = datetime.now()

    def to_dict(self):
        """转换为字典"""
        return {
            'error_type': self.error_type.value,
            'error_code': self.error_code,
            'message': self.message,
            'severity': self.severity.value,
            'details': self.details,
            'timestamp': self.timestamp.isoformat()
        }

class GlobalExceptionHandler:
    """全局异常处理器"""

    def __init__(self):
        self.error_logger = ErrorLogger()
        self.error_notifier = ErrorNotifier()
        self.error_recovery = ErrorRecovery()

    def handle_exception(self, exception: Exception, context: dict) -> ErrorResponse:
        """处理异常"""
        # 1. 识别错误类型
        error_info = self._classify_error(exception)

        # 2. 记录错误日志
        self.error_logger.log(error_info, context)

        # 3. 通知相关人员
        if error_info.severity in [ErrorSeverity.HIGH, ErrorSeverity.CRITICAL]:
            self.error_notifier.notify(error_info, context)

        # 4. 尝试恢复
        recovery_result = self.error_recovery.attempt_recovery(error_info, context)

        # 5. 返回错误响应
        return error_info

    def _classify_error(self, exception: Exception) -> ErrorResponse:
        """分类错误"""
        # 根据异常类型进行分类
        if isinstance(exception, ValueError):
            return ErrorResponse(
                error_type=ErrorType.VALIDATION_ERROR,
                error_code=ErrorCode.INVALID_CREDENTIALS.code,
                message=str(exception),
                severity=ErrorSeverity.LOW
            )
        elif isinstance(exception, PermissionError):
            return ErrorResponse(
                error_type=ErrorType.PERMISSION_ERROR,
                error_code=ErrorCode.USER_INACTIVE.code,
                message=str(exception),
                severity=ErrorSeverity.MEDIUM
            )
        elif isinstance(exception, TimeoutError):
            return ErrorResponse(
                error_type=ErrorType.TIMEOUT_ERROR,
                error_code=ErrorCode.LLM_TIMEOUT.code,
                message=str(exception),
                severity=ErrorSeverity.MEDIUM
            )
        elif isinstance(exception, ConnectionError):
            return ErrorResponse(
                error_type=ErrorType.NETWORK_ERROR,
                error_code=ErrorCode.API_CONNECTION_FAILED.code,
                message=str(exception),
                severity=ErrorSeverity.HIGH
            )
        else:
            return ErrorResponse(
                error_type=ErrorType.SYSTEM_ERROR,
                error_code=ErrorCode.AGENT_EXECUTION_FAILED.code,
                message=str(exception),
                severity=ErrorSeverity.HIGH
            )
```

### 3.4 错误恢复策略

```python
from typing import Optional
import time

class ErrorRecovery:
    """错误恢复器"""

    def __init__(self):
        self.recovery_strategies = {
            ErrorType.NETWORK_ERROR: self._recover_network_error,
            ErrorType.TIMEOUT_ERROR: self._recover_timeout_error,
            ErrorType.API_ERROR: self._recover_api_error,
            ErrorType.DATABASE_ERROR: self._recover_database_error,
        }

    def attempt_recovery(self, error_info: ErrorResponse, context: dict) -> Optional[bool]:
        """尝试恢复"""
        recovery_strategy = self.recovery_strategies.get(error_info.error_type)

        if recovery_strategy:
            return recovery_strategy(error_info, context)
        else:
            return False

    def _recover_network_error(self, error_info: ErrorResponse, context: dict) -> bool:
        """恢复网络错误"""
        # 重试机制
        max_retries = 3
        retry_delay = 5  # 秒

        for attempt in range(max_retries):
            try:
                # 重新执行操作
                time.sleep(retry_delay)
                return True
            except Exception as e:
                print(f"重试失败 (尝试 {attempt + 1}/{max_retries}): {e}")

        return False

    def _recover_timeout_error(self, error_info: ErrorResponse, context: dict) -> bool:
        """恢复超时错误"""
        # 增加超时时间并重试
        return False
```

```python
import openai
from openai import OpenAIError

class LLMErrorHandler:
    """LLM错误处理器"""

    def __init__(self, api_key: str, max_retries: int = 3):
        self.client = openai.OpenAI(api_key=api_key)
        self.max_retries = max_retries

    def call_llm(self, prompt: str, model: str = "gpt-4") -> str:

            try:
                response = self.client.chat.completions.create(
                    model=model,

                    print(f"LLM调用失败，重试 ({attempt + 1}/{self.max_retries}): {e}")

                else:
                    raise ErrorResponse(
                        error_type=ErrorType.LLM_ERROR,
                        error_code=ErrorCode.LLM_CALL_FAILED.code,
                        message=f"LLM调用失败: {str(e)}",
```

```python
import sqlite3
from sqlite3 import DatabaseError

class DatabaseErrorHandler:
    """数据库错误处理器"""

    def __init__(self, db_path: str):
        self.db_path = db_path
        self.connection = None

    def connect(self) -> sqlite3.Connection:
        """连接数据库"""
        try:
            self.connection = sqlite3.connect(self.db_path)
            self.connection.row_factory = sqlite3.Row
            return self.connection

        except sqlite3.Error as e:
            raise ErrorResponse(
                error_type=ErrorType.DATABASE_ERROR,
                error_code=ErrorCode.DATABASE_CONNECTION_FAILED.code,
                message=f"数据库连接失败: {str(e)}",
                severity=ErrorSeverity.CRITICAL
            )

    def execute_query(self, query: str, params: tuple = None) -> list:
        """执行查询"""
        try:
            cursor = self.connection.cursor()
            if params:
                cursor.execute(query, params)
            else:
                cursor.execute(query)
            return cursor.fetchall()

        except sqlite3.Error as e:
            raise ErrorResponse(
                error_type=ErrorType.DATABASE_ERROR,
                error_code=ErrorCode.DATABASE_QUERY_FAILED.code,
                message=f"数据库查询失败: {str(e)}",
                severity=ErrorSeverity.HIGH
            )

    def execute_insert(self, query: str, params: tuple = None) -> int:
        """执行插入"""
        try:
            cursor = self.connection.cursor()
            if params:
                cursor.execute(query, params)
            else:
                cursor.execute(query)
            self.connection.commit()
            return cursor.lastrowid

        except sqlite3.Error as e:
            self.connection.rollback()
            raise ErrorResponse(
                error_type=ErrorType.DATABASE_ERROR,
                error_code=ErrorCode.DATABASE_INSERT_FAILED.code,
                message=f"数据库插入失败: {str(e)}",
                severity=ErrorSeverity.HIGH
            )
```

### 4.3 API调用错误处理

```python
import requests
from requests.exceptions import RequestException

class APIErrorHandler:
    """API错误处理器"""

    def __init__(self, timeout: int = 30, max_retries: int = 3):
        self.timeout = timeout
        self.max_retries = max_retries

    def call_api(self, url: str, method: str = "GET", headers: dict = None,
                 data: dict = None, params: dict = None) -> dict:
        """调用API"""
        for attempt in range(self.max_retries):
            try:
                response = requests.request(
                    method=method,
                    url=url,
                    headers=headers,
                    json=data,
                    params=params,
                    timeout=self.timeout
                )
                response.raise_for_status()
                return response.json()

            except requests.exceptions.Timeout:
                if attempt < self.max_retries - 1:
                    print(f"API调用超时，重试 ({attempt + 1}/{self.max_retries})")
                    time.sleep(2 ** attempt)
                else:
                    raise ErrorResponse(
                        error_type=ErrorType.TIMEOUT_ERROR,
                        error_code=ErrorCode.API_TIMEOUT.code,
                        message="API调用超时",
                        severity=ErrorSeverity.MEDIUM
                    )

            except requests.exceptions.ConnectionError:
                if attempt < self.max_retries - 1:
                    print(f"API连接失败，重试 ({attempt + 1}/{self.max_retries})")
                    time.sleep(2 ** attempt)
                else:
                    raise ErrorResponse(
                        error_type=ErrorType.NETWORK_ERROR,
                        error_code=ErrorCode.API_CONNECTION_FAILED.code,
                        message="API连接失败",
                        severity=ErrorSeverity.HIGH
                    )

            except requests.exceptions.HTTPError as e:
                if e.response.status_code == 429:
                    raise ErrorResponse(
                        error_type=ErrorType.API_ERROR,
                        error_code=ErrorCode.API_RATE_LIMIT_EXCEEDED.code,
                        message="API调用频率超限",
                        severity=ErrorSeverity.MEDIUM
                    )
                elif e.response.status_code == 401:
                    raise ErrorResponse(
                        error_type=ErrorType.API_ERROR,
                        error_code=ErrorCode.INVALID_API_KEY.code,
                        message="无效的API密钥",
                        severity=ErrorSeverity.HIGH
                    )
                else:
                    raise ErrorResponse(
                        error_type=ErrorType.API_ERROR,
                        error_code=ErrorCode.API_RESPONSE_ERROR.code,
                        message=f"API响应错误: {str(e)}",
                        severity=ErrorSeverity.HIGH
                    )

            except RequestException as e:
                raise ErrorResponse(
                    error_type=ErrorType.API_ERROR,
                    error_code=ErrorCode.API_RESPONSE_ERROR.code,
                    message=f"API调用失败: {str(e)}",
```

## 5. 错误监控和报告

### 5.1 错误统计

```python
from collections import defaultdict
from datetime import datetime, timedelta

class ErrorMonitor:
    """错误监控器"""

    def __init__(self):
        self.error_counts = defaultdict(int)
        self.error_history = []

    def record_error(self, error_info: ErrorResponse):
        """记录错误"""
        # 统计错误次数
        self.error_counts[error_info.error_type] += 1

        # 记录错误历史
        self.error_history.append({
            'error_info': error_info,
            'timestamp': datetime.now()
        })

    def get_error_statistics(self, time_range: timedelta = timedelta(hours=24)) -> dict:
        """获取错误统计"""
        now = datetime.now()
        start_time = now - time_range

        # 过滤时间范围内的错误
        recent_errors = [
            error for error in self.error_history
            if error['timestamp'] >= start_time
        ]

        # 统计各类错误数量
        error_type_counts = defaultdict(int)
        for error in recent_errors:
            error_type_counts[error['error_info'].error_type] += 1

        return {
            'total_errors': len(recent_errors),
            'error_type_counts': {
                error_type.value: count
                for error_type, count in error_type_counts.items()
            },
            'time_range': f"{time_range.total_seconds() / 3600}小时"
        }

    def get_error_trend(self, error_type: ErrorType, days: int = 7) -> list:
        """获取错误趋势"""
        end_time = datetime.now()
        start_time = end_time - timedelta(days=days)

        # 按天统计错误数量
        daily_counts = defaultdict(int)
        for error in self.error_history:
            if (error['error_info'].error_type == error_type and
                start_time <= error['timestamp'] <= end_time):
                date = error['timestamp'].date()
                daily_counts[date] += 1

        # 生成趋势数据
        trend = []
        current_date = start_time.date()
        while current_date <= end_time.date():
            trend.append({
                'date': current_date.isoformat(),
                'count': daily_counts.get(current_date, 0)
            })
            current_date += timedelta(days=1)

        return trend
```

```python
生成时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}

"""
        # 获取各类错误的趋势
        for error_type in ErrorType:
            trend = self.error_monitor.get_error_trend(error_type, days=7)
            report += f"\n{error_type.value} 趋势:\n"
            for data in trend:
                report += f"  {data['date']}: {data['count']}次\n"

        return report
```
